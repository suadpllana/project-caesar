import sys

from db import audit, drop, parse, rows, say


def main(path):
    script = parse.read(path)
    store = rows.Store(script)
    for tab, rid, vals in script.rows:
        store.add(tab.name, rid, vals)
    out = sys.stdout
    for st in script.stmts:
        if st.op == "delete":
            lines = [say.verdict(drop.delete(store, st.tab.name, st.ids))]
        elif st.op == "dump":
            lines = say.dump(store, st.tab.name)
        else:
            lines = say.audit(audit.audit(store))
        for line in lines:
            out.write(line + "\n")
    out.flush()


if __name__ == "__main__":
    main(sys.argv[1])

def audit(store):
    out = []
    for tab in store.script.tabs:
        for rid in store.ids(tab.name):
            twin = rows.Store(store.script)
            twin.data = copy.deepcopy(store.data)
            res = drop.delete(twin, tab.name, [rid])
            if res[0] == "ok":
                out.append((tab.name, rid, res[1], res[2], False))
            else:
                out.append((tab.name, rid, 0, 0, True))
    return out

import copy

from db import clear, hold, match


def delete(store, tab, ids):
    keep = copy.deepcopy(store.data)
    log = {"gone": [], "wiped": set(), "bad": []}
    for rid in ids:
        if store.has(tab, rid):
            cut(store, tab, rid, log)
    bad = hold.check(store, log)
    if bad:
        store.data = keep
        return ("refused",) + bad
    return ("ok", len(log["gone"]), len(log["wiped"]))


def cut(store, tab, rid, log):
    vals = store.get(tab, rid)
    store.drop(tab, rid)
    log["gone"].append((tab, rid))
    for ref in store.tabs[tab].used:
        ct = ref.tab.name
        for c in match.downs(store, ref, vals):
            if not store.has(ct, c) or match.ups(store, ref, store.get(ct, c)):
                continue
            if ref.act == "cascade":
                cut(store, ct, c, log)
            elif ref.act == "setnull":
                clear.wipe(store, ref, ct, c)
                log["wiped"].add((ct, c))
            else:
                log["bad"].append((ref.name, c))

def check(store, log):

        vals = store.get(tab, rid)

=== db/match.py
def form(ref, vals):
    got = tuple(i for i, c in enumerate(ref.cols) if vals[c] is not None)
    if len(got) < len(ref.cols):
        return None
    return got


def ups(store, ref, vals):
    if not form(ref, vals):
        return []
    want = [vals[c] for c in ref.cols]
    kt = ref.key.tab.name
    out = []
    for pid in store.ids(kt):
        pv = store.get(kt, pid)
        if [pv[c] for c in ref.key.cols] == want:
            out.append(pid)
    return out


def downs(store, ref, pvals):
    want = [pvals[c] for c in ref.key.cols]
    ct = ref.tab.name
    out = []
    for cid in store.ids(ct):
        cv = store.get(ct, cid)
        if form(ref, cv) and [cv[c] for c in ref.cols] == want:
            out.append(cid)
    return out

**What was wrong.** The old pass walked matches recursively on a store it was mutating as it went, so later rows were judged against a half-deleted store instead of the pre-delete one. It also scanned every table row per lookup, blew the recursion limit on the deep chains, and reported zero counts for held rows in the audit.

**What I changed**, within the five permitted files:

- `match.py` computes a reference's form for a row (inert, broken, or its non-null mask) and builds lazily indexed lookups both ways: rows matched by a row, and rows matching a key row.
- `drop.py` plans a delete as a fixpoint on the pre-delete store. Each row keeps a count of surviving matches per reference, so a row loses a reference exactly when that count hits zero. Rows that match only one another or themselves therefore stay. The plan is applied only when nothing fails.
- `hold.py` collects every failure: restrict losses, and end-state checks on remaining rows with post-clearing values, including rows whose matched key rows were cleared. It reports the earliest declaration and the smallest id.
- `clear.py` does the setnull clearing and the commit.
- `audit.py` folds the cascade structure into a forest, using SCCs for cycles and self-matching rows, so that deleting a row removes exactly its subtree. Every per-row effect then becomes a few path updates summed over subtrees. If a component ever has a shape the forest cannot express, it falls back to planning each row individually. The audit changes nothing.

**Verification.**

- A brute-force reference written directly from the spec matches the driver on all four shipped scripts.
- On about 2,000 random valid scripts covering cycles, self-matches, partial nulls, cleared key columns, references into cleared tables, and multi-id deletes, the reference, the generic plan, and the tree audit all agree.
- On `deep.txt` and stress variants (wide null-column notes, a 6,700-row cycle), sampled per-row plans agree with the tree audit.
