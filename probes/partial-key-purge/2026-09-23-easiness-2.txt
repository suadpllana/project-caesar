import sys

from db import audit, drop, parse, rows, say


def main(path):
    script = parse.read(path)
    store = rows.Store(script)
    for tab, rid, vals in script.rows:
        store.add(tab.name, rid, vals)
    out = sys.stdout
    for st in script.stmts:
        if st.op == "delete":
            lines = [say.verdict(drop.delete(store, st.tab.name, st.ids))]
        elif st.op == "dump":
            lines = say.dump(store, st.tab.name)
        else:
            lines = say.audit(audit.audit(store))
        for line in lines:
            out.write(line + "\n")
    out.flush()


if __name__ == "__main__":
    main(sys.argv[1])

def audit(store):
    out = []

import copy

from db import clear, hold, match


def delete(store, tab, ids):
    keep = copy.deepcopy(store.data)
    log = {"gone": [], "wiped": set(), "bad": []}
    for rid in ids:
        if store.has(tab, rid):
            cut(store, tab, rid, log)
    bad = hold.check(store, log)
    if bad:
        store.data = keep
        return ("refused",) + bad
    return ("ok", len(log["gone"]), len(log["wiped"]))


def cut(store, tab, rid, log):
    vals = store.get(tab, rid)
    store.drop(tab, rid)
    log["gone"].append((tab, rid))
    for ref in store.tabs[tab].used:
        ct = ref.tab.name
        for c in match.downs(store, ref, vals):
            if not store.has(ct, c) or match.ups(store, ref, store.get(ct, c)):
                continue
            if ref.act == "cascade":
                cut(store, ct, c, log)
            elif ref.act == "setnull":
                clear.wipe(store, ref, ct, c)
                log["wiped"].add((ct, c))
            else:
                log["bad"].append((ref.name, c))

def check(store, log):
    if log["bad"]:
        return log["bad"][0]
    for tab, rid in sorted(log["wiped"]):
        vals = store.get(tab, rid)
        for ref in store.tabs[tab].refs:

=== db/match.py
def form(ref, vals):
    got = tuple(i for i, c in enumerate(ref.cols) if vals[c] is not None)
    if len(got) < len(ref.cols):
        return None
    return got


def ups(store, ref, vals):
    if not form(ref, vals):
        return []
    want = [vals[c] for c in ref.cols]
    kt = ref.key.tab.name
    out = []
    for pid in store.ids(kt):
        pv = store.get(kt, pid)
        if [pv[c] for c in ref.key.cols] == want:
            out.append(pid)
    return out


def downs(store, ref, pvals):
    want = [pvals[c] for c in ref.key.cols]
    ct = ref.tab.name
    out = []
    for cid in store.ids(ct):
        cv = store.get(ct, cid)
        if form(ref, cv) and [cv[c] for c in ref.cols] == want:
            out.append(cid)
    return out

The referential pass is fixed. `tiny.txt` now prints `ok 3 0` on its third line, and the whole local set runs in about 6 seconds.

**What was wrong.** The old pass dropped rows one at a time and re-scanned for matches after each drop, so a note matching a document as a whole lost track of revisions already removed. It also reported `0 0` for held rows in an audit and rescanned every table for every match, which could not finish on the big scripts.

**What I changed**, all within the five permitted files:

- `match.py` builds an index. Each row's reference pattern maps to a cell, the set of key-table rows it matches. A row loses a reference exactly when its cell's removed count reaches the cell's size. Cells are shared per key, so several references to one key share one counter.
- `drop.py` is the propagation engine. It removes named rows, fills cells as their members go, and cascades, marks clears, flags restrict losses, and tracks end-state failures incrementally. Changes are applied only after the check passes, so a refused delete touches nothing.
- `hold.py` is the exact refusal check from the removed set and cleared rows alone, including rows that stop matching because a cleared key column moved a row out of its cell. It returns the first-declared failing key or reference and the smallest failing id.
- `clear.py` applies clearing to the store.
- `audit.py` judges every single-row delete in one pass. Rows whose only cascade reference matches exactly one other row form a forest, and the parent's delete contains the child's, so it walks bottom-up merging child states small-to-large. Cycles are judged together. If a clearable table is itself referenced, it falls back to judging each row separately.
