def form(ref, vals):
    got = tuple(i for i, c in enumerate(ref.cols) if vals[c] is not None)
    if len(got) < len(ref.cols):
        return None
    return got


def ups(store, ref, vals):
    if not form(ref, vals):
        return []
    want = [vals[c] for c in ref.cols]
    kt = ref.key.tab.name
    out = []
    for pid in store.ids(kt):
        pv = store.get(kt, pid)
        if [pv[c] for c in ref.key.cols] == want:
            out.append(pid)
    return out


def downs(store, ref, pvals):
    want = [pvals[c] for c in ref.key.cols]
    ct = ref.tab.name
    out = []
    for cid in store.ids(ct):
        cv = store.get(ct, cid)
        if form(ref, cv) and [cv[c] for c in ref.cols] == want:
            out.append(cid)
    return out

import copy

from db import clear, hold, match


def delete(store, tab, ids):
    keep = copy.deepcopy(store.data)
    log = {"gone": [], "wiped": set(), "bad": []}
    for rid in ids:
        if store.has(tab, rid):
            cut(store, tab, rid, log)
    bad = hold.check(store, log)
    if bad:
        store.data = keep
        return ("refused",) + bad
    return ("ok", len(log["gone"]), len(log["wiped"]))


def cut(store, tab, rid, log):
    vals = store.get(tab, rid)
    store.drop(tab, rid)
    log["gone"].append((tab, rid))
    for ref in store.tabs[tab].used:
        ct = ref.tab.name
        for c in match.downs(store, ref, vals):
            if not store.has(ct, c) or match.ups(store, ref, store.get(ct, c)):
                continue
            if ref.act == "cascade":
                cut(store, ct, c, log)
            elif ref.act == "setnull":
                clear.wipe(store, ref, ct, c)
                log["wiped"].add((ct, c))
            else:
                log["bad"].append((ref.name, c))

class Store:
    def __init__(self, script):
        self.script = script
        self.tabs = {t.name: t for t in script.tabs}
        self.data = {t.name: {} for t in script.tabs}

    def add(self, tab, rid, vals):
        self.data[tab][rid] = list(vals)

    def has(self, tab, rid):
        return rid in self.data[tab]

    def get(self, tab, rid):
        return self.data[tab][rid]

    def ids(self, tab):
        return sorted(self.data[tab])

    def drop(self, tab, rid):
        del self.data[tab][rid]

    def put(self, tab, rid, col, val):
        self.data[tab][rid][col] = val

    if res[0] == "ok":

def dump(store, tab):
    out = []
    for rid in store.ids(tab):
        vals = ["-" if v is None else v for v in store.get(tab, rid)]
        out.append(" ".join([tab, str(rid)] + vals))
    return out


def audit(rows):
    out = []
    for tab, rid, gone, wiped, held in rows:
        out.append("%s %d %d %d %s" % (tab, rid, gone, wiped, "held" if held else "ok"))
    return out

from db import match
 
RESTRICT = match.RESTRICT
SETNULL = match.SETNULL
form = match.form
 
 
def judge(g, removed, lost):
    """Return (fails, newvals).
 
    fails maps declaration position -> smallest failing row id.
    newvals maps node -> its values after clearing (rows that are cleared).
    """
    fails = {}
    node_id = g.node_id
 
    def fail(pos, node):
        rid = node_id[node]
        if pos not in fails or rid < fails[pos]:
            fails[pos] = rid
 
    wiped = {}
    affected = set()
    for r, ri in lost:
        R = g.refs[ri]
        if R.act == RESTRICT:
            fail(R.pos, r)
            continue
        if r in removed:
            continue
        affected.add(r)
        if R.act == SETNULL:
            wiped.setdefault(r, set()).update(R.wipe)
 
    newvals = {}
    for r, w in wiped.items():
        v = list(g.vals[r])
        for c in w:
            v[c] = None
        newvals[r] = v
        t = g.node_tab[r]
        for K in t.keys:
            if any(v[c] is None for c in K.cols):
                fail(K.pos, r)
        for gid in g.node_groups[r]:
            K, mask = g.gkey[gid]
            if any(K.cols[i] in w for i in mask):
                for r2, _ in g.gdeps[gid]:
                    if r2 not in removed:
                        affected.add(r2)
 
    def gone(p, K, mask):
        if p in removed:
            return True
        w = wiped.get(p)
        return bool(w) and any(K.cols[i] in w for i in mask)
 
    for r in affected:
        v = newvals.get(r) or g.vals[r]
        for R in g.node_tab[r].refs:
            f = form(R, v)
            if f is None:
                continue
            if f is False:
                fail(R.pos, r)
                continue
            K = R.key
            mem = g.members(K, f, tuple(v[R.cols[i]] for i in f))
            if not mem or all(gone(p, K, f) for p in mem):
                fail(R.pos, r)
    return fails, newvals

**Result on tiny.txt**: the third line is now `ok 3 0`, and all sample scripts match an independent brute-force reference I wrote straight from the spec.

**What was wrong.** The old pass rescanned the whole key table per row, treated a partially-null `partial` reference as inert, and removed rows greedily as it walked, so rows that should stay when they only match one another, or that match many rows through a null column, were misjudged. Its audit also deep-copied the store per row, which is quadratic.

**What I built.**
- `match.py` builds a match graph once per store state: every row that is not inert for a reference belongs to exactly one group (key, non-null mask, values), and each group knows its size and its dependents. Post-clear masks are pre-registered so setnull outcomes can be judged without rescans.
- `drop.py` computes the exact least fixpoint of removal with per-group counters, then `hold.py` judges the end state (restrict losses, key nulls, broken rows, references left with no remaining match, including matches lost to nulled key columns) and reports the earliest declaration and smallest id. `clear.py` applies an accepted delete.
- `audit.py` exploits that single-row closures form a laminar family. Seeds are processed dependents-first via an iterative Tarjan SCC pass, and each seed's aggregate (removed count, cleared count, open group counters, failures) is merged wholesale into the closures that contain it. Rows whose contributions cannot be aggregated safely fall back to the exact engine, so correctness never depends on the fast path.

**Verification.**
- Reference agreement on tiny, loops and docs.
- Fuzzing against the reference with three generators: 2000 small sample-schema scripts, 120 mid-sized chain-heavy scripts with cycles and self-matches, and 1800 fully random schemas. All pass. The fuzzing caught and I fixed one real bug in the fast path: a row that can never be forced out could inherit failures recorded about itself from merged sub-closures. Such seeds now use the exact engine.
- On deep.txt, the fast audit agrees with the exact engine for all 36,067 rows.
