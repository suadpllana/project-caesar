class St:
    __slots__ = ("off", "vh", "foot")

    def __init__(self, vh):
        self.off = 0
        self.vh = vh
        self.foot = False


def settle(gm, st, cfg, held):
    gi, b = band.band(gm, st.off)
    w0, w1 = win.bounds(gm, st.off, st.vh, cfg.over)
    was = gm.top(held[0])
    m = win.sweep(gm, w0, w1)
    if st.foot:
        st.off = move.foot(gm.total(), st.vh)
    else:
        st.off = move.clamp(st.off + gm.top(held[0]) - was, gm.total(), st.vh)
    return gi, b, w0, w1, m, 1


def play(cfg, doc, evs, out):

    st = St(cfg.vh)

        out.frame(i, st.off, gm.ggid(gi), b, w0, w1, held[1], held[2], m, p)

class Geom:
    __slots__ = ("doc",)

    def __init__(self, doc):
        self.doc = doc

    def ngroups(self):
        return len(self.doc.gs)

    def ggid(self, gi):
        return self.doc.gs[gi].gid

    def ghh(self, gi):
        return self.doc.gs[gi].hh

    def gtop(self, gi):
        y = 0
        for g in self.doc.gs[:gi]:
            y += g.hh
            for rid in g.rows:
                y += self.doc.rh(g, rid)
        return y

    def count(self):
        i = 0
        for g in self.doc.gs:
            i += 1 + len(g.rows)
        return i

    def total(self):
        y = 0
        for g in self.doc.gs:
            y += g.hh
            for rid in g.rows:
                y += self.doc.rh(g, rid)
        return y

    def _place(self, i):
        for g in self.doc.gs:
            if i == 0:
                return g, -1
            i -= 1
            if i < len(g.rows):
                return g, i
            i -= len(g.rows)
        return None, -1

    def top(self, i):
        y = 0
        for g in self.doc.gs:
            if i == 0:
                return y
            y += g.hh
            i -= 1
            if i < len(g.rows):
                for rid in g.rows[:i]:
                    y += self.doc.rh(g, rid)
                return y
            for rid in g.rows:
                y += self.doc.rh(g, rid)
            i -= len(g.rows)
        return y

    def at(self, y):
        i = 0
        run = 0
        for g in self.doc.gs:
            if run + g.hh > y:
                return i
            run += g.hh
            i += 1
            for rid in g.rows:
                h = self.doc.rh(g, rid)
                if run + h > y:
                    return i
                run += h
                i += 1
        return i - 1

    def key(self, i):
        g, k = self._place(i)
        if k < 0:
            return "H%d" % g.gid
        return "R%d" % g.rows[k]

    def mark(self, i):
        g, k = self._place(i)
        if k < 0:
            return 0
        return self.doc.mark(g, g.rows[k])

    def ins(self, gid, pos, n):
        g = self.doc.byid[gid]
        g.rows[pos:pos] = self.doc.fresh(g, n)

    def dele(self, gid, pos, n):
        g = self.doc.byid[gid]
        del g.rows[pos:pos + n]

==== pane/move.py
def foot(total, vh):
    f = total - vh
    return f if f > 0 else 0


def clamp(off, total, vh):
    f = foot(total, vh)
    if off < 0:
        return 0
    if off > f:
        return f
    return off


def apply(gm, st, ev):
    st.foot = st.off == foot(gm.total(), st.vh)
    kind = ev[0]
    if kind == "scroll":
        st.off += ev[1]
    elif kind == "go":
        st.off = ev[1]
    elif kind == "size":
        st.vh = ev[1]
    st.off = clamp(st.off, gm.total(), st.vh)

Frame 1 of `tiny.txt` now reads exactly `f 1 s 52 g 2 b 18 w 4 8 h R4 -6 m 0 p 1`, and I hand-checked frame 0 against the spec as well. Both large runs finish in well under a second.

**What was wrong** in the rewritten half:

- The hold was taken at the offset, not at the anchor line (offset plus band), so it held the first visible item rather than the item across the line.
- The band was always the full header height. It never shrank when the next group's header pushed the pinned one off.
- The window only extended K items below the viewport, never above, and the last visible item was computed at offset plus viewport rather than the last pixel inside it.
- Foot-following was decided before the movement was applied instead of after.
- Settling ran exactly one pass, and moved the offset by the held item's displacement instead of placing it at top minus gap minus band.
- On a delete that removed the held item, the hold was simply retaken at the offset, with no transfer to the surviving neighbour and no gap adjustment.
- Every geometry query walked the whole document, which would not have scaled.

**What I changed**, within the six permitted files:

- `pane/geom.py` now keeps Fenwick trees over group heights, group item counts and per-group row heights, so tops, item lookup and pinned-group lookup are logarithmic. Measuring updates the trees in place. Inserts and deletes rebuild only the affected group's tree.
- `pane/band.py` computes the band as the smaller of the header height and the distance to the next header top, with the total standing in for the last group.
- `pane/win.py` clips K items on both sides and uses the last pixel of the viewport for the last visible item.
- `pane/hold.py` holds an item by group and row id, and on a removing delete moves to the first survivor after it, or the last before it, shifting the gap by the difference of their pre-edit tops.
- `pane/move.py` sets the foot flag after the movement and clamp.
- `pane/frame.py` runs up to P passes in the spec'd order and reports the last pass's band, group and window.
