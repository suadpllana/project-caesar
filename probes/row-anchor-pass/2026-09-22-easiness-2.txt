from pane import frame, say, spec, src


def run(text):
    cfg, decls, evs = spec.parse(text)
    doc = src.Doc(decls)
    out = say.Out()
    frame.play(cfg, doc, evs, out)
    return out.lines


def main(argv):
    if len(argv) != 2:
        sys.stderr.write("usage: run_pane.py <event-file>\n")
        return 2

        text = fh.read()
    try:
        lines = run(text)

    return 0

def settle(gm, st, cfg, held):
    gi, b = band.band(gm, st.off)
    w0, w1 = win.bounds(gm, st.off, st.vh, cfg.over)
    was = gm.top(held[0])
    m = win.sweep(gm, w0, w1)
    if st.foot:
        st.off = move.foot(gm.total(), st.vh)
    else:
        st.off = move.clamp(st.off + gm.top(held[0]) - was, gm.total(), st.vh)
    return gi, b, w0, w1, m, 1

        out.frame(i, st.off, gm.ggid(gi), b, w0, w1, held[1], held[2], m, p)

class Geom:
    __slots__ = ("doc",)

    def __init__(self, doc):
        self.doc = doc

    def ngroups(self):
        return len(self.doc.gs)

    def ggid(self, gi):
        return self.doc.gs[gi].gid

    def ghh(self, gi):
        return self.doc.gs[gi].hh

    def gtop(self, gi):
        y = 0
        for g in self.doc.gs[:gi]:
            y += g.hh
            for rid in g.rows:
                y += self.doc.rh(g, rid)
        return y

    def count(self):
        i = 0
        for g in self.doc.gs:
            i += 1 + len(g.rows)
        return i

    def total(self):
        y = 0
        for g in self.doc.gs:
            y += g.hh
            for rid in g.rows:
                y += self.doc.rh(g, rid)
        return y

    def _place(self, i):
        for g in self.doc.gs:
            if i == 0:
                return g, -1
            i -= 1
            if i < len(g.rows):
                return g, i
            i -= len(g.rows)
        return None, -1

    def top(self, i):
        y = 0
        for g in self.doc.gs:
            if i == 0:
                return y
            y += g.hh
            i -= 1
            if i < len(g.rows):
                for rid in g.rows[:i]:
                    y += self.doc.rh(g, rid)
                return y
            for rid in g.rows:
                y += self.doc.rh(g, rid)
            i -= len(g.rows)
        return y

    def at(self, y):
        i = 0
        run = 0
        for g in self.doc.gs:
            if run + g.hh > y:
                return i
            run += g.hh
            i += 1
            for rid in g.rows:
                h = self.doc.rh(g, rid)
                if run + h > y:
                    return i
                run += h
                i += 1
        return i - 1

    def key(self, i):
        g, k = self._place(i)
        if k < 0:
            return "H%d" % g.gid
        return "R%d" % g.rows[k]

    def mark(self, i):
        g, k = self._place(i)
        if k < 0:
            return 0
        return self.doc.mark(g, g.rows[k])

    def ins(self, gid, pos, n):
        g = self.doc.byid[gid]
        g.rows[pos:pos] = self.doc.fresh(g, n)

    def dele(self, gid, pos, n):
        g = self.doc.byid[gid]
        del g.rows[pos:pos + n]

def foot(total, vh):
    f = total - vh
    return f if f > 0 else 0


def clamp(off, total, vh):
    f = foot(total, vh)
    if off < 0:
        return 0
    if off > f:
        return f
    return off


def apply(gm, st, ev):
    st.foot = st.off == foot(gm.total(), st.vh)
    kind = ev[0]
    if kind == "scroll":
        st.off += ev[1]
    elif kind == "go":
        st.off = ev[1]
    elif kind == "size":
        st.vh = ev[1]
    st.off = clamp(st.off, gm.total(), st.vh)

Done. Both required lines now come out right, and the whole thing is fast.

**What was wrong.** The rewrite took the hold at the offset instead of the anchor line, used the pinned header's full height as the band without letting the next header push it off, built the window from the first visible item with no overscan above, ran a single pass, and decided whether the pane was following the foot before the movement rather than after. Every geometry query also walked the whole document, which would have been far too slow on the big files.

**What I changed.** Only the six permitted files under `pane/`.

- `geom.py` now keeps a Fenwick tree of row heights per group and one over group totals, so tops, lookups by position and measurements are logarithmic. Inserts and deletes rebuild only the edited group.
- `band.py` takes the smaller of the pinned header height and the distance to the next header top, with the total standing in for the last group.
- `win.py` runs from K before the first visible item to K after the last, clipped to the flow.
- `hold.py` tracks the held item by identity, takes it at the anchor line, and on a delete that removes it moves to the first survivor after, or the last before, adjusting the gap by the difference of the pre-edit tops.
- `move.py` clamps after the movement and sets the foot flag from the offset the movement left.
- `frame.py` runs up to P passes and stops once a pass measures nothing and leaves the offset in place.

**Verification.** Frame 1 of `evs/tiny.txt` prints `f 1 s 52 g 2 b 18 w 4 8 h R4 -6 m 0 p 1`. I also wrote a naive list-based oracle of the spec in `/tmp` and compared it against the pane on 3000 random generated documents with scrolls, jumps, resizes, inserts and deletes. There were zero mismatches.

One judgment call worth knowing: when a delete leaves the stale offset past the new total, nothing is visible during the first pass. I treat the last item of the flow as the visible item for the window in that pass. The offset is clamped at the end of the pass so later passes behave normally.
