def run(launch):
    blocks = [load.Blk(n) for n in range(launch.grid)]
    for b in blocks:
        b.busy = 0
        b.wait = None
    mem = Mem(launch.mem)
    pl = Place(launch, mem)
    turns = [Turn(launch.slots) for _ in range(launch.sms)]
    t = 0
    gone = []
    while True:
        for b in gone:
            pl.free(b)
        gone = []
        pl.fill(blocks, t)
        live = pl.live()
        if not live and pl.next == launch.grid:
            return blocks, None, 0, mem.gm
        ran = False
        for s in range(launch.sms):
            b = turns[s].pick(pl.rows[s], t)
            if b is None:
                continue
            ran = True
            hit = step(launch, mem, b, t)
            if b.end is not None:
                gone.append(b)
            if hit is not None:
                for w in live:
                    if w.wait == hit:
                        w.wait = None
        if not ran:
            later = [b.busy for b in live if b.busy > t]
            if not later:
                return blocks, t, launch.grid - pl.next, mem.gm
            t = min(later)
            continue
        t += 1
class Lines:
    def __init__(self, cap):
        self.cap = cap
        self.rows = {}

    def get(self, ln):
        row = self.rows.pop(ln, None)
        if row is not None:
            self.rows[ln] = row
        return row

    def put(self, ln, row):
        if ln not in self.rows and len(self.rows) >= self.cap:
            del self.rows[next(iter(self.rows))]
        self.rows.pop(ln, None)
        self.rows[ln] = row

    def has(self, ln):
        return ln in self.rows

def reg(tok):
    if len(tok) == 2 and tok[0] == "r" and tok[1] in "01234567":
        return int(tok[1])
    raise ValueError("bad register %r" % tok)


def opnd(tok):
    if tok in SPECIAL:
        return (tok, None)
    if tok.startswith("r"):
        return ("r", reg(tok))
    return ("k", int(tok))


def addr(tok):
    if not (tok.startswith("[") and tok.endswith("]")):
        raise ValueError("bad address %r" % tok)
    body = tok[1:-1]
    if body.startswith("r"):
        base, _, off = body.partition("+")
        return (reg(base), int(off) if off else 0)
    return (None, int(body))


def val(launch, blk, x):
    kind, v = x
    if kind == "r":
        return blk.reg[v]
    if kind == "k":
        return v
    if kind == "%bid":
        return blk.n
    if kind == "%sm":
        return blk.sm
    return launch.grid


def ea(blk, at):
    r, off = at
    return (blk.reg[r] if r is not None else 0) + off


def decode(f, labels):
    op, x = f[0], f[1:]
    if op == "mov":
        return Ins(op, rd=reg(x[0]), a=opnd(x[1]))
    if op in ("add", "sub", "mul", "slt"):
        return Ins(op, rd=reg(x[0]), a=opnd(x[1]), b=opnd(x[2]))
    if op == "mod":
        k = int(x[2])
        if k <= 0:
            raise ValueError("mod by %d" % k)
        return Ins(op, rd=reg(x[0]), a=opnd(x[1]), b=("k", k))
    if op in ("ld.ca", "ld.cg"):
        return Ins(op, rd=reg(x[0]), at=addr(x[1]))
    if op == "st":
        return Ins(op, at=addr(x[0]), a=opnd(x[1]))
    if op == "atom.add":
        return Ins(op, rd=reg(x[0]), at=addr(x[1]), a=opnd(x[2]))
    if op == "fence" or op == "exit":
        return Ins(op)
    if op in ("spin.ca", "spin.cg"):
        rd, at, b = reg(x[0]), addr(x[1]), opnd(x[3])
        if x[2] not in TESTS:
            raise ValueError("bad test %r" % x[2])
        if at[0] == rd or b == ("r", rd):
            raise ValueError("spin reads its own destination")
        return Ins(op, rd=rd, at=at, cmp=x[2], b=b)
    if op == "work" or op == "out":
        return Ins(op, a=opnd(x[0]))
    if op == "bra":
        return Ins(op, to=labels[x[0]])
    if op in ("brz", "brnz"):
        return Ins(op, a=opnd(x[0]), to=labels[x[1]])
    raise ValueError("bad op %r" % op)

from sim import load
from sim.mem import LW, Mem
from sim.place import Place
from sim.step import SPINS, step
from sim.turn import Turn

INF = float("inf")


def run(launch):
    blocks = [load.Blk(n) for n in range(launch.grid)]
    for b in blocks:
        b.busy = 0
    code = launch.code
    S = launch.sms
    sms = range(S)
    mem = Mem(launch.mem, S, launch.lines)
    pl = Place(launch)
    turns = [Turn(launch.slots) for _ in range(S)]
    rows = pl.rows
    t = 0
    gone = []
    exited = 0

    # While every ready block on a multiprocessor sits at a spin and the words
    # those spins read stay put, that multiprocessor evolves on its own and
    # periodically.  Once a state repeats we freeze it: it stops issuing, and
    # is caught up (replaying the remainder of the period) when a block on it
    # wakes or lands, or when one of its words is written.
    frozen = [None] * S  # (cycle frozen at, period) or None
    nxt = [INF] * S  # for a frozen multiprocessor, the first wake of a busy block
    seen = [{} for _ in sms]  # state signature -> cycle, while observing
    key = [None] * S  # ready blocks being observed
    deps = [set() for _ in sms]
    depmap = mem.depmap
    quiet_since = None
    nactive = S
    min_nxt = INF
    K = launch.slots
    isspin = [ins.op in SPINS for ins in code]

    def active_frozen_busy():
        return [s for s in sms if frozen[s] is not None and nxt[s] != INF]

    def clear_deps(s):
        for w in deps[s]:
            depmap[w].discard(s)
            if not depmap[w]:
                del depmap[w]
        deps[s] = set()

    def add_deps(s, ready):
        for b in ready:
            ins = code[b.pc]
            a = load.ea(b, ins.at)
            words = (a,) if ins.op == "spin.cg" else range(a - a % LW, a - a % LW + LW)
            for w in words:
                deps[s].add(w)
                depmap.setdefault(w, set()).add(s)

    def replay(s, k):
        f = frozen[s][0]
        row, tr = rows[s], turns[s]
        for _ in range(k):
            b = tr.pick(row, f)
            step(launch, mem, b, f)

    def unfreeze(s, k):
        nonlocal nactive, min_nxt
        f, p = frozen[s]
        replay(s, k % p)
        frozen[s] = None
        nactive += 1
        nxt[s] = INF
        min_nxt = min(nxt)
        key[s] = None
        seen[s].clear()
        clear_deps(s)

    def freeze(s, p):
        nonlocal nactive, min_nxt
        frozen[s] = (t, p)
        nactive -= 1
        nxt[s] = min((b.busy for b in rows[s] if b is not None and b.end is None and b.busy > t), default=INF)
        if nxt[s] < min_nxt:
            min_nxt = nxt[s]

    def on_write(w, r):
        for s in list(depmap[w]):
            if frozen[s] is not None:
                unfreeze(s, t - frozen[s][0] + (1 if s < r else 0))
            else:
                seen[s].clear()

    def on_place(s):
        if frozen[s] is not None:
            unfreeze(s, t - frozen[s][0])
        else:
            seen[s].clear()

    mem.on_write = on_write
    pl.on_place = on_place

    while True:

            for b in gone:
                pl.free(b)
            gone = []
        if pl.nfree and pl.next < launch.grid:
            pl.fill(blocks, t)
        if min_nxt <= t:

            if frozen[s] is not None:

            tr = turns[s]
            ready = []
            allspin = True

                if b is not None and b.end is None:

The model now produces the required seven lines for the example, and it agrees with an independent brute-force reference on every supplied small launch and on 4200 random launches.

**What was wrong.** The cycle model had drifted from the spec in several places at once:

- Caches were per block instead of per multiprocessor, and evicted by recency instead of fill order.
- A store updated every cache holding the line, `ld.cg` never dropped its line, and `fence` did nothing.
- A block at a spin was marked not ready after one failed attempt, so it never retried and never touched the cache again.
- Placement went round-robin instead of to the multiprocessor with the most free slots.
- Hang detection only looked at whether a store had landed, ignoring that spin attempts keep changing cache state.

**What I changed.** All six permitted files were rewritten:

- `sim/line.py` and `sim/mem.py` implement per-multiprocessor FIFO caches with the exact spec semantics for `ld.ca`, `ld.cg`, `st`, `atom.add` and `fence`.
- `sim/place.py` picks the most-free multiprocessor with lower-number tiebreak, and tracks free-slot counts.
- `sim/turn.py` and `sim/step.py` treat spinning blocks as ready and report spin successes to the clock.
- `sim/clock.py` runs the three-step cycle. For speed, a multiprocessor whose ready blocks are all spinning is watched until its state (rotation slot plus cache contents) repeats, then frozen. It is caught up by replaying the remainder of its period when a block on it wakes or lands, or when any word its spins read is written. A hang is declared at the first quiet cycle once every multiprocessor is frozen with no busy blocks.
