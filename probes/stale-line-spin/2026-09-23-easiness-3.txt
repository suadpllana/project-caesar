def run(launch):
    blocks = [load.Blk(n) for n in range(launch.grid)]
    for b in blocks:
        b.busy = 0
        b.wait = None
    mem = Mem(launch.mem)
    pl = Place(launch, mem)
    turns = [Turn(launch.slots) for _ in range(launch.sms)]
    t = 0
    gone = []
    while True:
        for b in gone:
            pl.free(b)
        gone = []
        pl.fill(blocks, t)
        live = pl.live()
        if not live and pl.next == launch.grid:
            return blocks, None, 0, mem.gm
        ran = False
        for s in range(launch.sms):
            b = turns[s].pick(pl.rows[s], t)
            if b is None:
                continue
            ran = True
            hit = step(launch, mem, b, t)
            if b.end is not None:
                gone.append(b)
            if hit is not None:
                for w in live:
                    if w.wait == hit:
                        w.wait = None
        if not ran:
            later = [b.busy for b in live if b.busy > t]
            if not later:
                return blocks, t, launch.grid - pl.next, mem.gm
            t = min(later)
            continue
        t += 1
class Lines:
    def __init__(self, cap):
        self.cap = cap
        self.rows = {}

    def get(self, ln):
        row = self.rows.pop(ln, None)
        if row is not None:
            self.rows[ln] = row
        return row

    def put(self, ln, row):
        if ln not in self.rows and len(self.rows) >= self.cap:
            del self.rows[next(iter(self.rows))]
        self.rows.pop(ln, None)
        self.rows[ln] = row

    def has(self, ln):
        return ln in self.rows

def reg(tok):
    if len(tok) == 2 and tok[0] == "r" and tok[1] in "01234567":
        return int(tok[1])
    raise ValueError("bad register %r" % tok)


def opnd(tok):
    if tok in SPECIAL:
        return (tok, None)
    if tok.startswith("r"):
        return ("r", reg(tok))
    return ("k", int(tok))


def addr(tok):
    if not (tok.startswith("[") and tok.endswith("]")):
        raise ValueError("bad address %r" % tok)
    body = tok[1:-1]
    if body.startswith("r"):
        base, _, off = body.partition("+")
        return (reg(base), int(off) if off else 0)
    return (None, int(body))


def val(launch, blk, x):
    kind, v = x
    if kind == "r":
        return blk.reg[v]
    if kind == "k":
        return v
    if kind == "%bid":
        return blk.n
    if kind == "%sm":
        return blk.sm
    return launch.grid


def ea(blk, at):
    r, off = at
    return (blk.reg[r] if r is not None else 0) + off


def decode(f, labels):
    op, x = f[0], f[1:]
    if op == "mov":
        return Ins(op, rd=reg(x[0]), a=opnd(x[1]))
    if op in ("add", "sub", "mul", "slt"):
        return Ins(op, rd=reg(x[0]), a=opnd(x[1]), b=opnd(x[2]))
    if op == "mod":
        k = int(x[2])
        if k <= 0:
            raise ValueError("mod by %d" % k)
        return Ins(op, rd=reg(x[0]), a=opnd(x[1]), b=("k", k))
    if op in ("ld.ca", "ld.cg"):
        return Ins(op, rd=reg(x[0]), at=addr(x[1]))
    if op == "st":
        return Ins(op, at=addr(x[0]), a=opnd(x[1]))
    if op == "atom.add":
        return Ins(op, rd=reg(x[0]), at=addr(x[1]), a=opnd(x[2]))
    if op == "fence" or op == "exit":
        return Ins(op)
    if op in ("spin.ca", "spin.cg"):
        rd, at, b = reg(x[0]), addr(x[1]), opnd(x[3])
        if x[2] not in TESTS:
            raise ValueError("bad test %r" % x[2])
        if at[0] == rd or b == ("r", rd):
            raise ValueError("spin reads its own destination")
        return Ins(op, rd=rd, at=at, cmp=x[2], b=b)
    if op == "work" or op == "out":
        return Ins(op, a=opnd(x[0]))
    if op == "bra":
        return Ins(op, to=labels[x[0]])
    if op in ("brz", "brnz"):
        return Ins(op, a=opnd(x[0]), to=labels[x[1]])
    raise ValueError("bad op %r" % op)

The model now produces the required seven lines for the example, and it agrees with an independent naive reference on every launch I could check.

**What was wrong.** The split had scattered several semantic errors across the modules:

- **Caches belonged to blocks, not multiprocessors.** Each placed block got its own cache, and every store wrote through to all caches, so stale reads never happened.
- **Replacement was least-recently-used** rather than dropping the line filled earliest.
- **Bypassing loads did not drop the line**, and `fence` did nothing.
- **Placement was round-robin** instead of the multiprocessor with the most free slots.
- **Failed spins parked the block** until a store hit the same address, instead of retrying every cycle in the rotation. That alone shifted the example's hang from 6 to 7.
- **Hang detection** only noticed "nobody issued", so a launch where spinners keep issuing forever never ended correctly.

**What I did.** I rewrote the six permitted modules against the spec:

- `sim/line.py` and `sim/mem.py`: one fill-order cache per multiprocessor with the exact hit, fill, drop, write-update, atomic and fence rules.
- `sim/place.py` and `sim/turn.py`: most-free-slots placement and slot rotation that remembers the slot.
- `sim/step.py`: the program is compiled once into one closure per instruction.
- `sim/clock.py`: the cycle loop. When every ready block sits at a spin, global memory is frozen, so each multiprocessor's cache-plus-rotation state is tracked until it repeats. With busy blocks outstanding the clock then jumps to the next wake-up, replaying only the remainder of each multiprocessor's period. With none outstanding it reports a hang. The hang cycle is the later of the phase start and the last wake-up, since a block leaving `work` can land directly on a spin.

**Verification.** I wrote a slow cycle-by-cycle reference straight from the spec and compared it against the model on about 6000 random launches plus 800 shrunk variants of the real patterns with swapped spin and load flavours. All agree. The seven small launches in `/app/launches` also match the reference.

**Timing** on this machine, single process:

| lookback_deep as shipped | 1.5 s |
| splitk_wide as shipped | 2.7 s |
